<script>

  // WKLEJ SWÓJ ADRES Z KROKU 1 PONIŻEJ:

  const APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxoCIgwvGGCqoUe5xkarvRHOPqRTFoD6wNbhghR_U96FWkpWqbTg1UZrWh3gxUd-dPjrQ/exec";


  function isVotingOpen() {

    const today = new Date();

    const lastDayOfMonth = new Date(today.getFullYear(), today.getMonth() + 1, 0).getDate();

    const currentDay = today.getDate();

    return currentDay >= (lastDayOfMonth - 6);

  }


  function setupVotingWindow() {

    const alertDiv = document.getElementById('statusAlert');

    const submitBtn = document.getElementById('submitBtn');

    

    if (!isVotingOpen()) {

      alertDiv.className = "alert alert-info";

      alertDiv.style.display = "block";

      alertDiv.innerText = "⏳ Typowanie jest obecnie zamknięte. Formularz otwiera się automatycznie w ostatnich 7 dniach każdego miesiąca.";

      submitBtn.disabled = true;

    }

  }


  document.getElementById('voteForm').addEventListener('submit', async function(e) {

    e.preventDefault();

    

    const alertDiv = document.getElementById('statusAlert');

    const submitBtn = document.getElementById('submitBtn');

    

    const participant = document.getElementById('participant').value;

    const ticker = document.getElementById('ticker').value;


    submitBtn.disabled = true;

    alertDiv.className = "alert alert-info";

    alertDiv.style.display = "block";

    alertDiv.innerText = "Wysyłanie zgłoszenia...";


    try {

      const response = await fetch(APPS_SCRIPT_URL, {

        method: 'POST',

        headers: { 'Content-Type': 'text/plain' },

        body: JSON.stringify({ participant, ticker })

      });

      

      const result = await response.json();

      

      if (result.status === 'success') {

        alertDiv.className = "alert alert-success";

        alertDiv.innerText = "✅ Typowanie zostało zapisane w arkuszu!";

        document.getElementById('voteForm').reset();

      } else {

        alertDiv.className = "alert alert-error";

        alertDiv.innerText = "❌ Błąd: " + result.message;

      }

    } catch (err) {

      alertDiv.className = "alert alert-error";

      alertDiv.innerText = "❌ Problem z połączeniem. Spróbuj ponownie.";

    } finally {

      if (isVotingOpen()) submitBtn.disabled = false;

    }

  });


  async function loadLeaderboard() {

    const tbody = document.getElementById('leaderboardBody');

    try {

      const res = await fetch(APPS_SCRIPT_URL);

      const data = await res.json();

      

      tbody.innerHTML = "";

      if (data.length === 0) {

        tbody.innerHTML = '<tr><td colspan="4" style="text-align:center;">Brak wyników do wyświetlenia.</td></tr>';

        return;

      }


      data.forEach((row, index) => {

        const percentage = (row.totalReturn * 100).toFixed(2);

        const badgeClass = row.totalReturn >= 0 ? "badge-positive" : "badge-negative";

        const sign = row.totalReturn >= 0 ? "+" : "";


        const tr = document.createElement('tr');

        tr.innerHTML = `

          <td><strong>#${index + 1}</strong></td>

          <td>${row.participant}</td>

          <td>${row.count}</td>

          <td class="${badgeClass}">${sign}${percentage}%</td>

        `;

        tbody.appendChild(tr);

      });

    } catch (err) {

      tbody.innerHTML = '<tr><td colspan="4" style="text-align:center;">Nie udało się pobrać aktualnych wyników.</td></tr>';

    }

  }


  setupVotingWindow();

  loadLeaderboard();
